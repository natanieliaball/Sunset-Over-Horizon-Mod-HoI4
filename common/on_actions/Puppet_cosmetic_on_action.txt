on_actions = {
    on_daily = {
        effect = {
            if = {
                limit = {
                    is_puppet = yes
                    NOT = { has_country_flag = puppet_flag }
                }
                set_country_flag = puppet_flag
            }
        }
    }
}

on_actions = {
    on_daily = {
        effect = {
            if = {
                limit = {
                    is_puppet = no
                    has_country_flag = puppet_flag
                }
                clr_country_flag = puppet_flag
                set_cosmetic_tag = Free_country
            }
        }
    }
}
