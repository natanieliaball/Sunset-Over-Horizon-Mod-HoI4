on_actions = {
    on_daily = {
        effect = {
            if = {
                limit = {
                    tag = SPT
                    has_government = fascism
                    has_country_leader = {
                        character = SPT_Meristra
                    }
                    date > 1936.2.13
                    NOT = { has_country_flag = SPT_meristra_dead }
                }
                set_country_flag = SPT_meristra_dead
                country_event = { id = CEC.1 days = 1 }

                    news_event = { id = news_nori.1 days = 1 } 
                      
            }
        }
    }
}
