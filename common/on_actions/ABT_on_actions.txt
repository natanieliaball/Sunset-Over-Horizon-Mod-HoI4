on_actions = {
    on_daily = {
        effect = {
            if = {
                limit = {
                    tag = SCH
                    is_puppet = yes
                    OVERLORD = {
                        OR = {
                            tag = ABT
                            original_tag = ABT
                        }
                        has_civil_war = yes
                    }
                    NOT = { has_country_flag = SCH_independence_triggered }
                }
                set_country_flag = SCH_independence_triggered
                country_event = { id = SCH.1 days = 3 }
            }
        }
    }
}
