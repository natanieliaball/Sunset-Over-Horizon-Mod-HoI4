# Anschluss FRK
add_namespace = NGC_anschluss_FRK

country_event = {
    id = NGC_anschluss_FRK.1
    title = NGC_anschluss_FRK.1.t
    desc = NGC_anschluss_FRK.1.d
    picture = GFX_report_event_generic_sign_treaty1
    is_triggered_only = yes

    option = {
        name = NGC_anschluss_FRK.1.a
        ai_chance = { factor = 60 }
        NGC = {
            annex_country = {
                target = FRK
                transfer_troops = yes
            }
            every_state = {
                limit = { is_core_of = FRK }
                add_core_of = NGC
            }
        }
    }

    option = {
        name = NGC_anschluss_FRK.1.b
        ai_chance = { factor = 40 }
        NGC = {
            create_wargoal = {
                type = annex_everything
                target = FRK
            }
        }
    }
}
