# Union démocratique avec ATL
country_event = {
    id = ABT_democracy.3
    title = ABT_democracy.3.t
    desc = ABT_democracy.3.d
    picture = GFX_report_event_generic_sign_treaty1
    is_triggered_only = yes

    option = {
        name = ABT_democracy.3.a
        ai_chance = { factor = 70 }
        ABT = {
            annex_country = {
                target = ATL
                transfer_troops = yes
            }
            every_state = {
                limit = { is_core_of = ATL }
                add_core_of = ABT
            }
            set_cosmetic_tag = AAU
        }
    }

    option = {
        name = ABT_democracy.3.b
        ai_chance = { factor = 30 }
    }
}
